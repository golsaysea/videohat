核心业务模块开关 opt_layout = QHBoxLayout() self.chk_tts = QCheckBox("启用 Edge-TTS 批量配音") self.chk_4grid = QCheckBox("开启四宫格分屏布局") opt_layout.addWidget(self.chk_tts) opt_layout.addWidget(self.chk_4grid) layout.addLayout(opt_layout) self.btn_select = QPushButton("1. 导入素材") self.btn_select.clicked.connect(self.select_videos) layout.addWidget(self.btn_select) self.btn_process = QPushButton("2. 执行自动化队列") self.btn_process.clicked.connect(self.start_processing) self.btn_process.setEnabled(False) layout.addWidget(self.btn_process) self.log_box = QTextEdit() self.log_box.setReadOnly(True) self.log_box.setStyleSheet("background-color: #1e1e1e; color: #00ff00; font-family: Consolas;") layout.addWidget(self.log_box) self.setLayout(layout) def select_videos(self): files, _ = QFileDialog.getOpenFileNames(self, "选择视频素材", "", "Video Files (*.mp4 *.mov)") if files: self.input_videos = files self.info_label.setText(f"已加载 {len(files)} 个素材文件。") self.btn_process.setEnabled(True) def start_processing(self): if not self.input_videos: return self.btn_select.setEnabled(False) self.btn_process.setEnabled(False) self.log_box.clear() self.worker = AutomationWorker( self.input_videos, "output_dir", self.chk_tts.isChecked(), self.chk_4grid.isChecked() ) self.worker.log_signal.connect(self.update_log) self.worker.finished_signal.connect(self.processing_finished) self.worker.start() def update_log(self, text): self.log_box.append(text) bar = self.log_box.verticalScrollBar() bar.setValue(bar.maximum()) def processing_finished(self, success): self.btn_select.setEnabled(True) self.btn_process.setEnabled(True) if __name__ == '__main__': app = QApplication(sys.argv) ex = VideoWorkshopApp() ex.show() sys.exit(app.exec_())
